cmake_minimum_required(VERSION 3.8)
project(vendor_robot_sdk_vendor)
find_package(ament_cmake REQUIRED)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
  message(FATAL_ERROR "The supplied vendor SDK is x86_64-only")
endif()
install(DIRECTORY include/ DESTINATION include/${PROJECT_NAME})
install(DIRECTORY lib/ DESTINATION lib)
add_executable(movex src/movex.cpp)
target_compile_features(movex PRIVATE cxx_std_17)
target_compile_options(movex PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(movex PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(movex PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/lib/libcr_sdk.so)
set_target_properties(movex PROPERTIES INSTALL_RPATH "$ORIGIN/../..")
install(TARGETS movex DESTINATION lib/${PROJECT_NAME})
ament_export_include_directories(include/${PROJECT_NAME})
ament_export_libraries(cr_sdk)
ament_package(CONFIG_EXTRAS "vendor_robot_sdk_vendor-extras.cmake")
